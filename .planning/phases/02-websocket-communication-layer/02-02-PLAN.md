- for { select {
    case c := <-h.register:
        h.mu.Lock()
        if len(h.clients) >= h.limit {
            h.mu.Unlock()
            _ = c.Close(websocket.CloseTryAgainLater, "server full")  // D-04 容量拒绝
            h.logger.Info("hub rejected client at capacity", zap.Int("limit", h.limit))
            continue
        }
        h.clients[c] = true
        h.mu.Unlock()
        h.logger.Info("hub registered client", zap.Int("total", len(h.clients)))
    case c := <-h.unregister:
        h.mu.Lock()
        if _, ok := h.clients[c]; ok {
            delete(h.clients, c)
            h.mu.Unlock()
            close(c.send)  // 信号 writePump 退出
            h.logger.Info("hub unregistered client", zap.Int("total", len(h.clients)))
        } else {
            h.mu.Unlock()
        }
    case <-ctx.Done():
        return  // D-12 资源清理
  } }

- defer func() { if r := recover(); r != nil { c.logger.Error("readPump panic recovered", zap.Any("recovered", r), zap.Stack("stack")) } }()
- for { select {
    default:
      msgType, data, err := c.conn.ReadMessage()
      if err != nil { return }  // 连接关闭或错误
      _ = msgType
      header, payload, err := DecodeFrame(data)（Plan 01）
      if err != nil {
        c.logger.Warn("frame decode failed", zap.Error(err))
        continue  // 跳过坏帧，不退出（避免恶意帧踢掉连接）
      }
      if err := c.handler.Route(ctx, c, header, payload); err != nil {
        c.logger.Warn("handler route failed", zap.Error(err))
      }
    case <-ctx.Done():
      return
  } }
- 注意：gorilla/websocket 的 ReadMessage 阻塞读取，select default 在初次进入循环时立即走 default；ctx 取消时由另一 goroutine 调用 c.conn.Close() 触发 ReadMessage 返回 error 实现退出 —— 本任务在 writePump 退出时 Close 连接以联动 readPump

实现 (c *Client) writePump(ctx context.Context)（Pattern 3 第 327-339 行 + D-12）：
- defer func() { _ = c.conn.Close() }()（触发 readPump 退出）
- for { select {
    case msg, ok := <-c.send:
      if !ok { return }  // channel closed by Hub on unregister
      if err := c.conn.WriteMessage(websocket.BinaryMessage, msg); err != nil { return }
    case <-ctx.Done():
      return
  } }

实现 (c *Client) Close(code websocket.StatusCode, reason string) error（hub.go 中骨架的完整实现）：
- c.closeOnce.Do(func() { _ = c.conn.WriteControl(websocket.CloseMessage, websocket.FormatCloseMessage(code, reason), time.Now().Add(time.Second)); _ = c.conn.Close() })
- return nil

实现 Heartbeat(ctx context.Context, c *websocket.Conn, interval time.Duration, logger *zap.Logger)（RESEARCH.md 第 423-441 行 + D-14）：
- ticker := time.NewTicker(interval); defer ticker.Stop()
- for { select {
    case <-ctx.Done(): return
    case <-ticker.C:
      deadline := time.Now().Add(interval / 2)
      if err := c.WriteControl(websocket.PingMessage, nil, deadline); err != nil {
        logger.Debug("heartbeat ping failed", zap.Error(err))
        return  // 连接已断，readPump 将通过 Hub.unregister 检测
