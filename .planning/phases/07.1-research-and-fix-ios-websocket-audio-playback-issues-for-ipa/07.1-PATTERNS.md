```javascript
// RESEARCH Ex.3 lines 414-444 — schedule on word_complete
function handleStatusFrame(payload) {
  const text = decodeText(payload);
  let data; try { data = JSON.parse(text); } catch (e) { /* keep L993-998 */ return; }
  if (data.type !== 'word_complete') { addLog('[STATUS] ' + text, 'status'); return; }

  const chunks = pendingPcm.get(data.word_id);
  pendingPcm.delete(data.word_id);
  if (!chunks || chunks.length === 0) return;

  // Merge PCM chunks (test-interface.html:1103-1111 shows the merge idiom)
  const totalLen = chunks.reduce((s, c) => s + c.length, 0);
  const pcmBytes = new Uint8Array(totalLen);
  let off = 0;
  for (const c of chunks) { pcmBytes.set(c, off); off += c.length; }

  // PCM Int16 → Float32 AudioBuffer (RESEARCH Pattern 3)
  const numSamples = pcmBytes.length / 2;
  const buf = audioCtx.createBuffer(1, numSamples, audioCtx.sampleRate);  // NOT hardcoded 16000
  const view = new DataView(pcmBytes.buffer, pcmBytes.byteOffset, pcmBytes.byteLength);
  const chan = buf.getChannelData(0);
  for (let i = 0; i < numSamples; i++) chan[i] = view.getInt16(i * 2, true) / 32768;

  // Gapless scheduling (RESEARCH Pattern 1)
  const src = audioCtx.createBufferSource();
  src.buffer = buf;
  src.connect(audioCtx.destination);
  if (nextStartTime < audioCtx.currentTime) nextStartTime = audioCtx.currentTime; // underrun
  src.start(nextStartTime);
  nextStartTime += buf.duration;
}
```
